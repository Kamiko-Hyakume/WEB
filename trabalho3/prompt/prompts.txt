Função salvarGrupo():
js$scope.salvarGrupo = function() {
  if (!$scope.formGrupo.nome || !$scope.formGrupo.roboId) return;
  $scope.gruposEmpresa.push({
    id: 'g' + Date.now(),
    roboId: $scope.formGrupo.roboId,
    nome: $scope.formGrupo.nome,
    empresaIds: angular.copy($scope.formGrupo.empresaIds)
  });
  // Retornar ao modal de agendamento, pré-selecionando o grupo recém-criado
  $scope.formAg.grupoId = $scope.gruposEmpresa[$scope.gruposEmpresa.length - 1].id;
  $scope.modalType = 'agendamento';
};

4. Função helper — getGrupoSelecionado()
js$scope.getGrupoSelecionado = function() {
  return $scope.gruposEmpresa.find(function(g) { return g.id === $scope.formAg.grupoId; }) || { empresaIds: [] };
};

5. Adaptar salvarAgendamento() para batch
Quando formAg.modoExecucao === 'grupo', iterar sobre as empresas do grupo e criar uma entrada de agendamento por empresa, todas com o mesmo batchId para rastreamento:
js$scope.salvarAgendamento = function() {
  if ($scope.formAg.modoExecucao === 'grupo') {
    var grupo = $scope.getGrupoSelecionado();
    var batchId = 'batch-' + Date.now();
    grupo.empresaIds.forEach(function(eId) {
      $scope.agendamentos.push({
        empresa: $scope.getEmpresaNome(eId),
        robo: $scope.getRoboNome($scope.formAg.robo),
        desc: '[Lote] ' + grupo.nome + ' — ' + $scope.formAg.hora,
        cron: $scope.formAg.cronExpr,
        proxima: '09/06 ' + $scope.formAg.hora,
        ativo: true,
        batchId: batchId,
        grupoNome: grupo.nome
      });
    });
  } else {
    // lógica individual existente, sem alteração
  }
  $scope.modalOpen = false;
};
Na tabela de agendamentos (page === 'agendamentos'), quando a.batchId existir, exibir uma badge "Lote" ao lado do nome do robô e mostrar a.grupoNome na coluna Empresa em vez do nome individual.
